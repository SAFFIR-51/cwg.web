function tabKey(event, index, count, select, refs) {
  const offset = event.key === 'ArrowRight' ? 1 : event.key === 'ArrowLeft' ? -1 : 0;
  if (!offset && !['Home', 'End'].includes(event.key)) return;
  event.preventDefault();
  const next = event.key === 'Home' ? 0 : event.key === 'End' ? count - 1 : (index + offset + count) % count;
  select(next);
  refs.current[next]?.focus();
}

const BOARD=[7,12,23,4,18,31,9,26,42];
const PATTERNS=[[7,12,23],[4,18,31],[9,26,42]];
function MemoryGame() {
  const [phase,setPhase]=useState('idle');
  const [round,setRound]=useState(0);
  const [picked,setPicked]=useState([]);
  const [seconds,setSeconds]=useState(5);
  const target=PATTERNS[round % PATTERNS.length];
  useEffect(()=>{
    if(phase!=='reveal') return undefined;
    const id=setInterval(()=>setSeconds(n=>Math.max(0,n-1)),1000);
    return ()=>clearInterval(id);
  },[phase]);
  useEffect(()=>{if(phase==='reveal' && seconds===0) setPhase('guess');},[phase,seconds]);
  const start=()=>{setPicked([]);setSeconds(5);setPhase('reveal');};
  const choose=n=>{
    if(phase!=='guess') return;
    setPicked(prev=>prev.includes(n) ? prev.filter(v=>v!==n) : prev.length<3 ? [...prev,n] : prev);
  };
  const won=picked.length===3 && picked.every(n=>target.includes(n));
  return <div className="own-memory"><div className="memory-topline"><span>넘버 센스</span><small>웹 미니 체험</small></div><p className="memory-instruction" aria-live="polite">{phase==='idle' ? '숫자 세 개, 기억할 수 있나요?' : phase==='reveal' ? '파란 숫자 세 개를 기억해 보세요.' : phase==='guess' ? '방금 본 숫자 세 개를 골라주세요.' : won ? '기억했네요! 세 개 모두 맞았어요.' : '정답은 '+target.join(' · ')+'입니다.'}</p><div className="own-memory-board">{BOARD.map(n=><button type="button" key={n} aria-label={n+'번'+(phase==='reveal' && target.includes(n) ? ', 기억할 숫자' : phase==='done' && target.includes(n) ? ', 정답' : '')} aria-pressed={picked.includes(n)} disabled={phase!=='guess'} onClick={()=>choose(n)} className={(phase==='reveal' && target.includes(n) ? 'is-target ' : '')+(phase==='done' && target.includes(n) ? 'is-answer ' : '')+(picked.includes(n) ? 'is-picked' : '')}>{phase==='idle' ? <span aria-hidden="true">·</span> : n}</button>)}</div><div className="memory-controls">{phase==='idle' ? <button type="button" className="own-button" onClick={start}>5초 동안 기억하기<Icon name="play_arrow" size={18} /></button> : phase==='reveal' ? <><span className="memory-countdown">{seconds}초</span><button type="button" onClick={()=>setSeconds(10)} className="own-quiet-link">조금 더 볼게요</button><button type="button" onClick={()=>setPhase('guess')} className="own-quiet-link">준비됐어요</button></> : phase==='guess' ? <button type="button" className="own-button" disabled={picked.length!==3} onClick={()=>setPhase('done')}>정답 확인하기 <span>{picked.length}/3</span></button> : <button type="button" className="own-button" onClick={()=>{setRound(n=>n+1);start();}}>다른 숫자로 한 번 더<Icon name="refresh" size={18} /></button>}</div><small className="memory-disclaimer">앱에서는 1~45 숫자판으로 즐겨요. 웹 체험은 포인트가 적립되지 않아요.</small></div>;
}
